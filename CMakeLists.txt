cmake_minimum_required(VERSION 3.16)
project(BBMOCBS_pex_H LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
if(POLICY CMP0167)
  cmake_policy(SET CMP0167 NEW)
endif()
find_package(Boost REQUIRED COMPONENTS program_options)
file(GLOB_RECURSE SOLVER_SOURCES CONFIGURE_DEPENDS src/*.cpp)
list(REMOVE_ITEM SOLVER_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp)
add_library(bbmocbs_core STATIC ${SOLVER_SOURCES})
target_include_directories(bbmocbs_core PUBLIC inc ${Boost_INCLUDE_DIRS})
add_executable(bbmocbs_approx src/main.cpp)
target_link_libraries(bbmocbs_approx PRIVATE bbmocbs_core Boost::program_options)
option(BUILD_REGRESSION_CHECKS "Build algorithm regression checks" ON)
if(BUILD_REGRESSION_CHECKS)
  enable_testing()
  add_executable(pex_h_regression validation/pex_h_regression.cpp)
  target_link_libraries(pex_h_regression PRIVATE bbmocbs_core)
  add_test(NAME pex_h_regression COMMAND pex_h_regression)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME pex_h_cli COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/validation/smoke_cli.py $<TARGET_FILE:bbmocbs_approx>)
    add_test(NAME pex_comparison_checks COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/validation/comparison_checks.py)
  endif()
endif()
